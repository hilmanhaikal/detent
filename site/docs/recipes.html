A price range filter

Two thumbs on one track. RM 100, 250 and 500 pull harder, because that is where people stop.

RM 100 to RM 500
function PriceRange({ onChange }) {
  const box = useRef(null)
  useEffect(() => {
    const [low, high] = box.current.querySelectorAll('[role="slider"]')
    const marks = [100, 250, 500].map((at) => ({ at, strong: true }))
    const lo = detent(low, { min: 0, max: 450, step: 10, value: 100, marks, onChange: (v) => (hi.set({ min: v + 50 }), onChange([v, hi.value])) })
    const hi = detent(high, { min: 150, max: 1000, step: 10, value: 500, marks, onChange: (v) => (lo.set({ max: v - 50 }), onChange([lo.value, v])) })
    return () => (lo.destroy(), hi.destroy())
  }, [])
  return (
    <div ref={box} className="price">
      <div aria-label="Lowest price" />
      <div aria-label="Highest price" />
    </div>
  )
}

A booking time picker

Quarter hours from 08:00 to 21:00. Whole hours pull harder, and a flick still stops on a slot.

function TimePicker({ onChange }) {
  const ref = useRef(null)
  const [rows, setRows] = useState(46)
  useEffect(() => {
    const marks = Array.from({ length: 14 }, (_, h) => ({ at: h * 4, strong: true }))
    return detent(ref.current, { min: 0, max: 52, value: 46, marks, path: 'y', size: 44, coast: 0.3, text: time, onInput: setRows, onChange }).destroy
  }, [])
  return <div ref={ref} aria-label="Booking time">{draw(rows)}</div>
}

A bottom sheet with heights

Peek, half or full, never in between. Drag the handle, flick it, or tap it.

Your rider is 6 min away

2 × Chicken rice, 1 × Iced lemon tea

function OrderSheet() {
  const frame = useRef(null)
  const handle = useRef(null)
  useEffect(() => {
    const heights = [0.24, 0.56, 0.92]
    const sheet = detent(handle.current, {
      min: 0.24, max: 0.92, step: 0.01, marks: heights, snap: 'marks',
      path: 'y', size: () => frame.current.clientHeight,
      onInput: (h) => frame.current.style.setProperty('--h', h),
      onTap: (h) => sheet.go(heights[(heights.indexOf(h) + 1) % 3]),
    })
    return sheet.destroy
  }, [])
  return <div ref={frame} className="phone"><div className="sheet"><div ref={handle} aria-label="Order sheet height" /></div></div>
}

A tip dial

One click per percent on a RM 48 bill. The common tips, 10, 15 and 20, are one easy turn.

10% tip, RM 4.80. Total RM 52.80
function TipDial({ bill, onChange }) {
  const ref = useRef(null)
  useEffect(() => {
    const marks = [10, 15, 20].map((at) => ({ at, strong: true }))
    return detent(ref.current, {
      min: 0, max: 25, value: 10, path: 'turn', marks, reach: { reach: 0.3, strong: 1.4 },
      onInput: (v) => (ref.current.style.rotate = `${v * 10.8 - 135}deg`),
      onChange,
    }).destroy
  }, [])
  return <div ref={ref} aria-label="Tip" className="dial" />
}

A floating video in the corners

A small player that rests in a corner of the screen. Throw it to another corner, or use the arrow keys.

Video
function FloatingVideo({ src }) {
  const ref = useRef(null)
  useEffect(() => detent2d(ref.current, { points: 'corners', inset: 16, cssVar: ['--x', '--y'] }).destroy, [])
  return <video ref={ref} src={src} aria-label="Floating video" className="pip" />
}
