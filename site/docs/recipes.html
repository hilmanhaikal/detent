A price range filter

Two thumbs on one track. RM 100, 250 and 500 pull harder, because that is where people stop.

RM 100 to RM 500
function PriceRange({ onChange }) {
  const box = useRef(null)
  useEffect(() => {
    const [low, high] = box.current.querySelectorAll('[role="slider"]')
    const marks = [100, 250, 500].map((at) => ({ at, strong: true }))
    const lo = detent(low, { min: 0, max: 450, step: 10, value: 100, marks, onChange: (v) => (hi.set({ min: v + 50 }), onChange([v, hi.value])) })
    const hi = detent(high, { min: 150, max: 1000, step: 10, value: 500, marks, onChange: (v) => (lo.set({ max: v - 50 }), onChange([lo.value, v])) })
    return () => (lo.destroy(), hi.destroy())
  }, [])
  return (
    <div ref={box} className="price">
      <div aria-label="Lowest price" />
      <div aria-label="Highest price" />
    </div>
  )
}

A booking time picker

Quarter hours from 08:00 to 21:00. Whole hours pull harder, and a flick still stops on a slot.

function TimePicker({ onChange }) {
  const ref = useRef(null)
  const [rows, setRows] = useState(46)
  useEffect(() => {
    const marks = Array.from({ length: 14 }, (_, h) => ({ at: h * 4, strong: true }))
    return detent(ref.current, { min: 0, max: 52, value: 46, marks, path: 'y', size: 44, coast: 0.3, text: time, onInput: setRows, onChange }).destroy
  }, [])
  return <div ref={ref} aria-label="Booking time">{draw(rows)}</div>
}

A bottom sheet with heights

Peek, half or full, never in between. Drag the handle, flick it, or tap it.

Your rider is 6 min away

2 × Chicken rice, 1 × Iced lemon tea

function OrderSheet() {
  const frame = useRef(null)
  const handle = useRef(null)
  useEffect(() => {
    const heights = [0.24, 0.56, 0.92]
    const sheet = detent(handle.current, {
      min: 0.24, max: 0.92, step: 0.01, marks: heights, snap: 'marks',
      path: 'y', size: () => frame.current.clientHeight,
      onInput: (h) => frame.current.style.setProperty('--h', h),
      onTap: (h) => sheet.go(heights[(heights.indexOf(h) + 1) % 3]),
    })
    return sheet.destroy
  }, [])
  return <div ref={frame} className="phone"><div className="sheet"><div ref={handle} aria-label="Order sheet height" /></div></div>
}

A tip dial

One click per percent on a RM 48 bill. The common tips, 10, 15 and 20, are one easy turn.

10% tip, RM 4.80. Total RM 52.80
function TipDial({ bill, onChange }) {
  const ref = useRef(null)
  useEffect(() => {
    const marks = [10, 15, 20].map((at) => ({ at, strong: true }))
    return detent(ref.current, {
      min: 0, max: 25, value: 10, path: 'turn', marks, reach: { reach: 0.3, strong: 1.4 },
      onInput: (v) => (ref.current.style.rotate = `${v * 10.8 - 135}deg`),
      onChange,
    }).destroy
  }, [])
  return <div ref={ref} aria-label="Tip" className="dial" />
}

Swipe to dismiss a notification

Swipe it away either way, or press Dismiss. Undo brings it back the way it left.

Your parcel is out for deliveryArriving today between 2 and 4 pm.

Notification dismissed. 

function Notice({ text, onDismiss }) {
  const ref = useRef(null)
  const card = useRef(null)
  useEffect(() => {
    card.current = detentThrow(ref.current, { cssVar: ['--x', '--y', '--tilt'], tilt: 0, onThrow: onDismiss })
    return card.current.destroy
  }, [])
  return (
    <div ref={ref} className="notice">
      <p>{text}</p>
      <button type="button" onClick={() => card.current.throw('right')}>Dismiss</button>
    </div>
  )
}

Pull to refresh a list

Pull the orders down at the top of the list. Scrolled down, it is a plain list. Refresh does the same for keys and screen readers.

function Orders({ load }) {
  const ref = useRef(null)
  const pull = useRef(null)
  useEffect(() => {
    pull.current = detentPull(ref.current, { cssVar: '--pull', onPull: () => load().finally(() => pull.current.done()) })
    return pull.current.destroy
  }, [])
  return (
    <>
      <button type="button" onClick={() => pull.current.trigger()}>Refresh</button>
      <div ref={ref} className="orders">{/* rows */}</div>
    </>
  )
}

Hold to delete

Hold to delete a saved address. A tap opens a confirm step instead, so no one has to hold.

Hold to delete, or tap for a confirm step.

function DeleteAddress({ onDelete }) {
  const ref = useRef(null)
  const [step, setStep] = useState(false)
  useEffect(() => detentHold(ref.current, { cssVar: '--progress', onTap: () => setStep(true), onConfirm: onDelete }).destroy, [])
  return (
    <>
      <button ref={ref} type="button" aria-expanded={step} aria-describedby="delete-hint">Delete this address</button>
      <p id="delete-hint">Hold to delete, or tap for a confirm step.</p>
      {step && (
        <div role="group" aria-labelledby="delete-hint">
          <button type="button" autoFocus onClick={() => setStep(false)}>Cancel</button>
          <button type="button" onClick={onDelete}>Delete</button>
        </div>
      )}
    </>
  )
}

A card that flips

A word on the front, its meaning on the back. Drag it over, flick it, or press Flip. It always lands face up, never on its edge.

WordEphemeral
MeaningLasting a very short time.
.card { transform-style: preserve-3d; transform: rotateY(calc(var(--yaw) * 1deg)); }
.card > div { position: absolute; inset: 0; backface-visibility: hidden; }
.card .back { transform: rotateY(180deg); }
function FlashCard({ word, meaning }) {
  const ref = useRef(null)
  const card = useRef(null)
  useEffect(() => {
    const faces = [{ yaw: 0, name: `Word: ${word}` }, { yaw: 180, name: `Meaning: ${meaning}` }]
    card.current = detent3d(ref.current, { faces, axes: 'yaw', cssVar: ['--yaw', '--pitch'] })
    return card.current.destroy
  }, [word, meaning])
  return (
    <>
      <div ref={ref} aria-label="Flash card" className="frame">
        <div className="card" aria-hidden="true"><div>{word}</div><div className="back">{meaning}</div></div>
      </div>
      <button type="button" onClick={() => card.current.go(1 - card.current.index)}>Flip</button>
    </>
  )
}

A delivery tracker you scrub

A timeline moves the parcel from stop to stop, and a detent slider scrubs it. The stops are the timeline's labels, and marks() turns them into marks, so the slider settles on a stop and says its name.

  1. Ordered
  2. Packed
  3. Shipped
  4. Out for delivery
  5. Delivered
.route { --p: 0; }
.done { transform-origin: left; scale: var(--p) 1; }
.run { position: absolute; inset: 0; translate: calc(var(--p) * 100%) 0; }
function Tracker({ stops }) {
  const route = useRef(null)
  const scrub = useRef(null)
  useEffect(() => {
    const tl = timeline({ paused: true }).label(stops[0])
    stops.slice(1).forEach((name, i) => tl.add(route.current, { to: { '--p': (i + 1) / (stops.length - 1) }, duration: 700, ease: 'inOut' }).label(name))
    const marks = tl.marks()
    const control = detent(scrub.current, {
      min: 0, max: tl.duration, marks, snap: 'marks',
      text: (ms) => stops[marks.findIndex((m) => m.at === ms)],
      onInput: (ms) => tl.seek(ms),
    })
    return () => (control.destroy(), tl.destroy())
  }, [stops])
  return (
    <>
      <div ref={route} className="route">…</div>
      <div ref={scrub} aria-label="Delivery progress" className="slider" />
    </>
  )
}
